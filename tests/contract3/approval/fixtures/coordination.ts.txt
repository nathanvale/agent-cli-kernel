// Disposable test-owned coordination fixture, materialized into a temporary
// stamped layout at addons/state/coordination.ts. It is NOT the T3 lock
// protocol: mkdir is used only because it is atomic across local processes.
// It obtains a real kernel-issued ExclusiveScope through the state seam.
import { mkdirSync, rmdirSync, statSync } from "node:fs";
import { contractFor } from "../../kernel/index.ts";
import type { ExclusiveScope, StateIssuer } from "../../kernel/index.ts";

let issuer: StateIssuer | undefined;
export const state = contractFor.stateAddon((granted) => {
  issuer = granted;
  return {
    ifNoResult: granted.ifNoResult({ kind: "command", command: "fixture.pending", args: [] }),
    attach() {
      return {
        transaction() {
          return Promise.reject(new Error("Fixture transactions are not exercised."));
        },
      };
    },
  };
});

function code(error: unknown): string | undefined {
  return error instanceof Error && "code" in error && typeof error.code === "string" ? error.code : undefined;
}

export class FixtureLock {
  #held = false;
  constructor(readonly path: string) {}
  tryAcquire(): boolean {
    try {
      mkdirSync(this.path);
      this.#held = true;
      return true;
    } catch (error) {
      if (code(error) === "EEXIST") return false;
      throw error;
    }
  }
  async waitAcquire(boundMilliseconds: number): Promise<boolean> {
    const deadline = Date.now() + boundMilliseconds;
    while (Date.now() < deadline) {
      if (this.tryAcquire()) return true;
      await new Promise((resolve) => setTimeout(resolve, 5));
    }
    return false;
  }
  active(): boolean {
    if (!this.#held) return false;
    try {
      return statSync(this.path).isDirectory();
    } catch {
      return false;
    }
  }
  forget(): void {
    this.#held = false;
  }
  release(): void {
    if (this.#held) rmdirSync(this.path);
    this.#held = false;
  }
}

export function exclusiveScope(runId: string, scope: string, active: () => boolean): ExclusiveScope {
  if (!issuer) throw new Error("State fixture was not installed.");
  return issuer.exclusiveScope(runId, scope, active);
}
