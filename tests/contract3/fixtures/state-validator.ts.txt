function fixture() {
  const state = contractFor.stateAddon((issuer) => ({ ifNoResult: issuer.ifNoResult({kind: "command", command: "specimen.pending", args: []}), attach() { throw new Error("No state execution in validator fixture."); } }));
  const change = { commandIdentity: "specimen.change", route: ["change"], summary: "Change.", input, capabilities: {}, effectClass: "repository-local", effects: ["a", "b"], ifNoResult: state.ifNoResult, heldLockCause: "ROW_3", unresolvedCaseCause: "ROW_5", stations } as const;
  return contractFor({ cli: "specimen", commands: [change, recovery], causes, state, installedAddons: ["state"] });
}
function inventory(row: CauseRow): Effects {
  switch(row.transactionState) {
    case "unchanged": return empty;
    case "completed": return { ...empty, completed: ["a"] };
    case "partially-completed": return { ...empty, completed: ["a"], remaining: ["b"] };
    case "unknown": return { ...empty, uncertain: ["a"] };
  }
}
function specimen(index: number) {
  const selected = stations[index];
  if (!selected) throw new Error("Missing independent row.");
  const { guidance, ...row } = selected.row;
  void guidance;
  const result: WireResult = { ...row, commandIdentity: "specimen.change", causeCode: selected.causeCode, runId: "run-1", effectClass: "repository-local", data: null, repairAction: selected.repairAction, effects: inventory(selected.row), ...selected.guidance, ...(selected.row.retryable ? { retryDelayMilliseconds: 5 } : {}) };
  return {contractVersion: "3.0.0", message: "Observed row.", availablePaths: ["specimen.change"], result};
}
test("every truth-table arm validates its literal correlation and rejects a wrong exit", async () => {
  const contract = fixture();
  expect(cases).toHaveLength(27);
  for (let index = 0; index < cases.length; index++) {
    const value = specimen(index);
    expect(await contract.validate(value)).toMatchObject({ok: true});
    expect((await contract.validate({...value, result: {...value.result, exitCode: 99}})).ok).toBe(false);
  }
});
test("effect-state inventories reject wrong, overlapping, unsorted and incomplete claims", async () => {
  const contract = fixture();
  const invalid = [
    { index: 0, effects: {...empty, remaining: ["a"]} },
    { index: 1, effects: empty },
    { index: 21, effects: {...empty, completed: ["a"]} },
    { index: 24, effects: empty },
    { index: 24, effects: {...empty, completed: ["a"], uncertain: ["a"]} },
    { index: 24, effects: {...empty, uncertain: ["b", "a"]} },
  ];
  expect(invalid).toHaveLength(6);
  for (const {index, effects} of invalid) {
    const value = specimen(index);
    expect((await contract.validate({...value, result: {...value.result, effects}})).ok).toBe(false);
  }
  const unknown = specimen(24);
  expect((await contract.validate({...unknown, result: {...unknown.result, effects: {...empty, inventoryComplete: false}}})).ok).toBe(true);
});

const observedProofs: unknown[] = [];
function installation() {
  return contractFor.stateAddon((issuer) => ({
    ifNoResult: issuer.ifNoResult({kind: "command", command: "specimen.pending", args: []}),
    attach(runId, effects) {
      return { async transaction() {
        const [first, ...rest] = effects;
        if (!first) throw new Error("Fixture requires an effect.");
        const proof = issuer.transactionProof(runId, {transactionState: "completed", effects: {completed: [first, ...rest], remaining: [], uncertain: [], inventoryComplete: true}});
        observedProofs.push(proof);
        return proof;
      } };
    },
  }));
}
function executable() {
  const state = installation();
  const completed = station(commonCauses, {commandIdentity: "specimen.change", causeCode: "SUCCESS_COMPLETED", trigger: "Fixture completion.", guidance: {next: {kind: "done"}}, repairAction: null, retryDelayPolicy: {kind: "none"}, reachability: "required", unreachableRationale: null, independentObservable: "synthetic-state-custody"});
  const change = changing({capabilities: {}, identity: "specimen.change", route: ["change"], summary: "Synthetic change.", input, effectClass: "repository-local", effects: ["a"], ifNoResult: state.ifNoResult, heldLockCause: "ROW_3", unresolvedCaseCause: "ROW_5", stations: [...stations.filter((entry) => entry.causeCode === "ROW_3" || entry.causeCode === "ROW_5"), completed], run: async (_input, context) => context.result(completed, {message: "Completed fixture.", data: null, transaction: await context.state.transaction(async () => {})})});
  return {state, completed, contract: contractFor({cli: "specimen", commands: [change, recovery], causes, state, installedAddons: ["state"]})};
}
test("cloned descriptors and forged, cloned, foreign and superseded proofs are refused", async () => {
  const first = executable();
  expect(() => contractFor({cli: "specimen", commands: [], installedAddons: ["state"], state: {...first.state}})).toThrow();
  const run = await first.contract.execute("specimen.change", [], "same-run-text");
  expect(run.ok).toBe(true);
  if (!run.ok) throw new Error("Fixture execution refused.");
  const issued = observedProofs.at(-1);
  const facts = {runId: "same-run-text", message: "Completed synthetic fixture.", data: null, transaction: issued};
  expect(await Reflect.apply(first.contract.build, undefined, [first.completed, facts])).toMatchObject({ok: true});
  const second = executable();
  expect(await Reflect.apply(second.contract.build, undefined, [second.completed, facts])).toMatchObject({ok: false, path: "transaction"});
  for (const transaction of [{runId: "same-run-text", transactionState: "completed", effects: {...empty, completed: ["a"]}}, structuredClone(issued)]) expect(await Reflect.apply(first.contract.build, undefined, [first.completed, {...facts, transaction}])).toMatchObject({ok: false, path: "transaction"});
  expect(await Reflect.apply(first.contract.build, undefined, [first.completed, {...facts, runId: "other-run"}])).toMatchObject({ok: false, path: "transaction"});
  await first.contract.execute("specimen.change", [], "same-run-text");
  expect(await Reflect.apply(first.contract.build, undefined, [first.completed, facts])).toMatchObject({ok: false, path: "transaction"});
});
test("changing selected capabilities and declared approval precede handler entry", async () => {
  const fixture = executable();
  const events: string[] = [];
  const change = changing({identity: "specimen.authorised", route: ["authorised"], summary: "Admitted synthetic change.", input, effectClass: "repository-local", effects: ["a"], ifNoResult: fixture.state.ifNoResult, heldLockCause: "ROW_3", unresolvedCaseCause: "ROW_5", stations: [...stations.filter((entry) => entry.causeCode === "ROW_3" || entry.causeCode === "ROW_5").map((entry) => ({...entry, commandIdentity: "specimen.authorised" as const})), {...fixture.completed, commandIdentity: "specimen.authorised"}], capabilities: {programs: true, approval: true, diagnostics: true}, approval: () => ({authority: fixture.contract.boundary.authority("fixture-owner"), resource: fixture.contract.boundary.resource("fixture-resource"), resourceRevision: "revision-1", effects: ["a"]}), run: async (_input, context) => { events.push("handler"); context.diagnostics.record("started", null); return context.result({commandIdentity: "specimen.authorised", causeCode: "SUCCESS_COMPLETED"}, {message: "Approved fixture.", data: null, transaction: await context.state.transaction(async () => {})}); }});
  const contract = contractFor({cli: "specimen", commands: [change, recovery], causes, state: fixture.state, installedAddons: ["state", "programs", "approval", "diagnostics"]});
  const programs = {inspect() { throw new Error("Not exercised."); }, supplied() { throw new Error("Not exercised."); }, domain() { throw new Error("Not exercised."); }};
  const diagnostics = {record() { events.push("diagnostic"); }};
  expect(await contract.execute("specimen.authorised", [], "denied", {programs, diagnostics, approval: {async check() {events.push("approval-denied"); return false;}}})).toMatchObject({ok: false, path: "approval"});
  expect(events).toEqual(["approval-denied"]);
  events.length = 0;
  expect(await contract.execute("specimen.authorised", [], "admitted", {programs, diagnostics, approval: {async check() {events.push("approval-accepted"); return true;}}})).toMatchObject({ok: true});
  expect(events).toEqual(["approval-accepted", "handler", "diagnostic"]);
});
