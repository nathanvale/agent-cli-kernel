// Independent effect ledger: written by the fixture state add-on only when an
// effect's operation ran, before read-back is judged.
const ledger: { runId: string; effect: string; installation: string }[] = [];
const scopes = new Map<string, unknown>();
let scopeMode: "active" | "inactive" | "throw" = "active";
let workStarts = 0;
let doubleProof = false;
let doubleProofSecond = "not-run";
function installation(name: string) {
  return contractFor.stateAddon((issuer) => ({
    ifNoResult: issuer.ifNoResult({ kind: "command", command: "specimen.pending", args: [] }),
    attach(runId, effects) {
      return {
        async transaction(work) {
          workStarts++;
          scopes.set(`${name}:${runId}`, issuer.exclusiveScope(runId, "resource-1", () => { if (scopeMode === "throw") throw new Error("Coordination lost."); return scopeMode === "active"; }));
          const confirmed: string[] = [];
          await work({
            async record(effect, operation, readBack) {
              const out = await operation(issuer.recordedEffect(runId, effect, (_r, _e, start) => start()));
              ledger.push({ runId, effect, installation: name });
              if (await readBack(out)) confirmed.push(effect);
              return out;
            },
          });
          const observation = { transactionState: confirmed.length > 0 ? "completed" : "unchanged", effects: { completed: confirmed, remaining: [], uncertain: [], inventoryComplete: true } } as const;
          const proof = issuer.transactionProof(runId, observation);
          if (doubleProof) {
            try { issuer.transactionProof(runId, observation); doubleProofSecond = "issued"; } catch { doubleProofSecond = "refused"; }
          }
          void effects;
          return proof;
        },
      };
    },
  }));
}
type Mode = "second" | "overlap" | "double-proof" | "refuse-after-effect" | "refuse-after-unchanged" | "completed" | "scope";
let mode: Mode = "completed";
const observed: Record<string, unknown> = {};
function fixture(name: string) {
  const state = installation(name);
  const st = (causeCode: "SUCCESS_UNCHANGED" | "SUCCESS_COMPLETED") => station(commonCauses, { commandIdentity: "specimen.change", causeCode, trigger: `Fixture ${causeCode}.`, guidance: { next: { kind: "done" } }, repairAction: null, ...words });
  const held = (causeCode: "HELD" | "PRIOR") => station(table, { commandIdentity: "specimen.change", causeCode, trigger: `Fixture ${causeCode}.`, guidance: handoff, repairAction: "Inspect the lock.", ...words });
  const stations = [held("HELD"), held("PRIOR"), st("SUCCESS_UNCHANGED"), st("SUCCESS_COMPLETED")] as const;
  const change = changing({
    identity: "specimen.change", route: ["change"], summary: "Change.", input, capabilities: {}, effectClass: "repository-local", effects: ["e"], ifNoResult: state.ifNoResult, heldLockCause: "HELD", unresolvedCaseCause: "PRIOR", stations,
    run: async (_input, context) => {
      const nothing = async () => {};
      const recordE = async (tx: { record: (effect: "e", operation: () => Promise<string>, readBack: (value: string) => Promise<boolean>) => Promise<string> }) => { await tx.record("e", async () => "written", async () => true); };
      if (mode === "second") {
        const pA = await context.state.transaction(nothing);
        let second = "fulfilled";
        try { await context.state.transaction(recordE); } catch { second = "rejected"; }
        observed.second = second;
        return context.result({ commandIdentity: "specimen.change", causeCode: "SUCCESS_UNCHANGED" }, { message: "Unchanged.", data: null, transaction: pA });
      }
      if (mode === "overlap") {
        const slow = async (tx: Parameters<typeof recordE>[0]) => { await new Promise((resolve) => setTimeout(resolve, 20)); await recordE(tx); };
        const settled = await Promise.allSettled([context.state.transaction(slow), context.state.transaction(recordE)]);
        observed.overlap = settled.map((item) => item.status);
        const first = settled.find((item) => item.status === "fulfilled");
        if (!first || first.status !== "fulfilled") throw new Error("No fulfilled transaction.");
        return context.result({ commandIdentity: "specimen.change", causeCode: "SUCCESS_COMPLETED" }, { message: "Completed.", data: null, transaction: first.value });
      }
      if (mode === "refuse-after-effect" || mode === "refuse-after-unchanged") {
        await context.state.transaction(mode === "refuse-after-effect" ? recordE : nothing);
        return context.result({ commandIdentity: "specimen.change", causeCode: "HELD" }, { message: "Held.", data: null, effects: empty });
      }
      if (mode === "double-proof") {
        const proof = await context.state.transaction(nothing);
        return context.result({ commandIdentity: "specimen.change", causeCode: "SUCCESS_UNCHANGED" }, { message: "Unchanged.", data: null, transaction: proof });
      }
      const proof = await context.state.transaction(recordE);
      return context.result({ commandIdentity: "specimen.change", causeCode: "SUCCESS_COMPLETED" }, { message: "Completed.", data: null, transaction: proof });
    },
  });
  return { state, stations, contract: contractFor({ cli: "specimen", commands: [change, recovery], causes, state, installedAddons: ["state"] }) };
}

async function run(contract: ReturnType<typeof fixture>["contract"], selected: Mode, runId: string) {
  mode = selected;
  const before = ledger.length;
  const starts = workStarts;
  const executed = await contract.execute("specimen.change", [], runId);
  const built = executed.ok ? await contract.buildResult(executed.value) : executed;
  return {
    executed: executed.ok,
    built: built.ok ? { ok: true, causeCode: built.value.result.causeCode, completed: built.value.result.effects.completed } : { ok: false, path: built.path },
    ledger: ledger.slice(before).map((entry) => entry.effect),
    workStarts: workStarts - starts,
  };
}

// R02: a T4-style approval runtime. At 7de078c it trusts the brand, as T4 does;
// when the kernel exports issuedScope it also asks the kernel about issuance.
function approvalRuntime(runId: string, descriptor: unknown) {
  const consumed: string[] = [];
  return {
    consumed,
    async revalidateAndConsume(_request: unknown, coordination: { runId: string; scope: string; isActive: () => boolean }) {
      const issued = typeof issuedScope === "function" ? issuedScope(coordination, runId, descriptor) === true : true;
      if (!issued || coordination.isActive() !== true || coordination.runId !== runId || coordination.scope !== "resource-1") return { admitted: false };
      consumed.push(runId);
      return { admitted: true };
    },
  };
}

async function attempt(runId: string, coordination: unknown) {
  const runtime = approvalRuntime(runId, a.state);
  const result = await runtime.revalidateAndConsume({}, coordination as { runId: string; scope: string; isActive: () => boolean });
  return { admitted: result.admitted, consumed: runtime.consumed.length };
}
