// R01 (F13) and R02 (K2) observations. Materialised at
// starters/contract3/src/addons/state/state-repair.ts, linted there, and run by
// the tests-scope driver. It returns observations keyed by frozen case
// identity; the parent test owns every expected value.
import {
  changing,
  commonCauses,
  contractFor,
  issuedScope,
  station,
} from "../../kernel/index.ts";
import type {
  EffectTransaction,
  ExclusiveScope,
  StateAddonDescriptor,
  StateIssuer,
  StateObservation,
  StateFactoryResult,
} from "../../kernel/index.ts";
import {
  buildUnchecked,
  causes,
  coordination,
  empty,
  field,
  input,
  recovery,

} from "./repair-shared.ts";

type Mode =
  | "second"
  | "overlap"
  | "double-proof"
  | "refuse-after-effect"
  | "refuse-after-unchanged"
  | "completed";
type ScopeMode = "active" | "inactive" | "throw";

// Independent effect ledger: written by the fixture state add-on only when an
// effect's operation ran, before read-back is judged.
type World = {
  readonly ledger: { readonly runId: string; readonly effect: string }[];
  readonly scopes: Map<string, ExclusiveScope>;
  readonly issuers: Map<string, StateIssuer>;
  scopeMode: ScopeMode;
  workStarts: number;
  doubleProof: boolean;
  doubleProofSecond: string;
  mode: Mode;
  second: string;
  overlap: string[];
};
const world: World = {
  ledger: [],
  scopes: new Map(),
  issuers: new Map(),
  scopeMode: "active",
  workStarts: 0,
  doubleProof: false,
  doubleProofSecond: "not-run",
  mode: "completed",
  second: "not-run",
  overlap: [],
};

function installation(name: string): StateAddonDescriptor {
  return contractFor.stateAddon((issuer) => {
    world.issuers.set(name, issuer);
    return installer(name, issuer);
  });
}
function installer(name: string, issuer: StateIssuer): StateFactoryResult {
  return {
    ifNoResult: issuer.ifNoResult({
      kind: "command",
      command: "specimen.pending",
      args: [],
    }),
    attach<Effect extends string>(runId: string) {
      return {
        async transaction(
          work: (transaction: EffectTransaction<Effect>) => Promise<void>,
        ) {
          world.workStarts++;
          world.scopes.set(
            `${name}:${runId}`,
            issuer.exclusiveScope(runId, "resource-1", () => {
              if (world.scopeMode === "throw")
                throw new Error("Coordination lost.");
              return world.scopeMode === "active";
            }),
          );
          const confirmed: Effect[] = [];
          await work({
            async record(effect, operation, readBack) {
              const out = await operation(
                issuer.recordedEffect(runId, effect, (_run, _effect, start) =>
                  start(),
                ),
              );
              world.ledger.push({ runId, effect });
              if (await readBack(out)) confirmed.push(effect);
              return out;
            },
          });
          const [first, ...rest] = confirmed;
          const observation: StateObservation<Effect> =
            first === undefined
              ? { transactionState: "unchanged", effects: empty }
              : {
                  transactionState: "completed",
                  effects: {
                    completed: [first, ...rest],
                    remaining: [],
                    uncertain: [],
                    inventoryComplete: true,
                  },
                };
          const proof = issuer.transactionProof(runId, observation);
          if (world.doubleProof) {
            try {
              issuer.transactionProof(runId, observation);
              world.doubleProofSecond = "issued";
            } catch {
              world.doubleProofSecond = "refused";
            }
          }
          return proof;
        },
      };
    },
  };
}

const recordE = async (transaction: EffectTransaction<"e">) => {
  await transaction.record(
    "e",
    () => Promise.resolve("written"),
    () => Promise.resolve(true),
  );
};
const nothing = async () => {};
const done = { next: { kind: "done" } } as const;
function changeStations() {
  return [
    ...coordination("specimen.change"),
    station(commonCauses, {
      commandIdentity: "specimen.change",
      causeCode: "SUCCESS_UNCHANGED",
      trigger: "Fixture unchanged.",
      guidance: done,
      repairAction: null,
      ...words,
    }),
    station(commonCauses, {
      commandIdentity: "specimen.change",
      causeCode: "SUCCESS_COMPLETED",
      trigger: "Fixture completed.",
      guidance: done,
      repairAction: null,
      ...words,
    }),
  ];
}
function fixture(name: string) {
  const state = installation(name);
  const change = changing({
    identity: "specimen.change",
    route: ["change"],
    summary: "Change.",
    input,
    capabilities: {},
    effectClass: "repository-local",
    effects: ["e"],
    ifNoResult: state.ifNoResult,
    heldLockCause: "HELD",
    unresolvedCaseCause: "PRIOR",
    stations: changeStations(),
    run: async (_input, context) => {
      const unchanged = {
        commandIdentity: "specimen.change",
        causeCode: "SUCCESS_UNCHANGED",
      } as const;
      const completed = {
        commandIdentity: "specimen.change",
        causeCode: "SUCCESS_COMPLETED",
      } as const;
      if (world.mode === "second") {
        const first = await context.state.transaction(nothing);
        try {
          await context.state.transaction(recordE);
          world.second = "fulfilled";
        } catch {
          world.second = "rejected";
        }
        return context.result(unchanged, {
          message: "Unchanged.",
          data: null,
          transaction: first,
        });
      }
      if (world.mode === "overlap") {
        const slow = async (transaction: EffectTransaction<"e">) => {
          await new Promise((resolve) => setTimeout(resolve, 20));
          await recordE(transaction);
        };
        const settled = await Promise.allSettled([
          context.state.transaction(slow),
          context.state.transaction(recordE),
        ]);
        world.overlap = settled.map((item) => item.status).sort();
        const proof = settled.find((item) => item.status === "fulfilled");
        if (proof?.status !== "fulfilled")
          throw new Error("No fulfilled transaction.");
        return context.result(completed, {
          message: "Completed.",
          data: null,
          transaction: proof.value,
        });
      }
      if (
        world.mode === "refuse-after-effect" ||
        world.mode === "refuse-after-unchanged"
      ) {
        await context.state.transaction(
          world.mode === "refuse-after-effect" ? recordE : nothing,
        );
        return context.result(
          { commandIdentity: "specimen.change", causeCode: "HELD" },
          { message: "Held.", data: null, effects: empty },
        );
      }
      if (world.mode === "double-proof")
        return context.result(unchanged, {
          message: "Unchanged.",
          data: null,
          transaction: await context.state.transaction(nothing),
        });
      return context.result(completed, {
        message: "Completed.",
        data: null,
        transaction: await context.state.transaction(recordE),
      });
    },
  });
  return {
    state,
    contract: contractFor({
      cli: "specimen",
      commands: [change, recovery()],
      causes,
      state,
      installedAddons: ["state"],
    }),
  };
}
type Fixture = ReturnType<typeof fixture>;

async function run(contract: Fixture["contract"], mode: Mode, runId: string) {
  world.mode = mode;
  const before = world.ledger.length;
  const starts = world.workStarts;
  const executed = await contract.execute("specimen.change", [], runId);
  const built = executed.ok
    ? await contract.buildResult(executed.value)
    : executed;
  return {
    executed: executed.ok,
    built: built.ok
      ? {
          ok: true,
          causeCode: built.value.result.causeCode,
          completed: built.value.result.effects.completed,
        }
      : { ok: false, path: built.path },
    ledger: world.ledger.slice(before).map((entry) => entry.effect),
    workStarts: world.workStarts - starts,
  };
}

// Foreign proofs: a real proof is reachable only inside a handler, so capture
// one and present it for another run and to another installation's contract.
async function foreignProofs(other: Fixture["contract"]) {
  let held: unknown;
  const state = installation("C");
  const change = changing({
    identity: "specimen.change",
    route: ["change"],
    summary: "Change.",
    input,
    capabilities: {},
    effectClass: "repository-local",
    effects: ["e"],
    ifNoResult: state.ifNoResult,
    heldLockCause: "HELD",
    unresolvedCaseCause: "PRIOR",
    stations: changeStations(),
    run: async (_input, context) => {
      const transaction = await context.state.transaction(recordE);
      held = transaction;
      return context.result(
        { commandIdentity: "specimen.change", causeCode: "SUCCESS_COMPLETED" },
        { message: "Completed.", data: null, transaction },
      );
    },
  });
  const contract = contractFor({
    cli: "specimen",
    commands: [change, recovery()],
    causes,
    state,
    installedAddons: ["state"],
  });
  const executed = await contract.execute("specimen.change", [], "run-held");
  const selected = {
    commandIdentity: "specimen.change",
    causeCode: "SUCCESS_COMPLETED",
  };
  const facts = {
    runId: "run-held",
    message: "Completed.",
    data: null,
    transaction: held,
  };
  const verdict = async (target: object, value: object) => {
    const built = await buildUnchecked(target, selected, value);
    return field(built, "ok") === true ? "ok" : field(built, "path");
  };
  return {
    executed: executed.ok,
    own: (await verdict(contract, facts)) === "ok",
    otherRun: await verdict(contract, { ...facts, runId: "run-other" }),
    otherInstallation: await verdict(other, facts),
  };
}

// R02: a T4-style approval runtime that asks the kernel about issuance and
// then repeats the brand-level checks T4 makes.
function approvalRuntime(runId: string, descriptor: StateAddonDescriptor) {
  const consumed: string[] = [];
  return {
    consumed,
    revalidateAndConsume(coordination: unknown) {
      const isActive = field(coordination, "isActive");
      const active: unknown =
        typeof isActive === "function"
          ? Reflect.apply(isActive, coordination, [])
          : false;
      if (
        !issuedScope(coordination, runId, descriptor) ||
        active !== true ||
        field(coordination, "runId") !== runId ||
        field(coordination, "scope") !== "resource-1"
      )
        return { consumed: false };
      consumed.push(runId);
      return { consumed: true };
    },
  };
}

export async function observeStateRepair(): Promise<Record<string, unknown>> {
  const a = fixture("A");
  const b = fixture("B");
  const out: Record<string, unknown> = {};
  out["R01.second-transaction-refused"] = {
    ...(await run(a.contract, "second", "run-second")),
    second: world.second,
  };
  out["R01.overlapping-transaction-refused"] = {
    ...(await run(a.contract, "overlap", "run-overlap")),
    overlap: world.overlap,
  };
  world.doubleProof = true;
  out["R01.single-proof-per-attachment"] = {
    ...(await run(a.contract, "double-proof", "run-double")),
    secondProof: world.doubleProofSecond,
  };
  world.doubleProof = false;
  out["R01.refusal-after-effect-refused"] = await run(
    a.contract,
    "refuse-after-effect",
    "run-refuse-effect",
  );
  out["R01.refusal-after-unchanged-admitted"] = await run(
    a.contract,
    "refuse-after-unchanged",
    "run-refuse-unchanged",
  );
  out["R01.completed-twin"] = await run(a.contract, "completed", "run-completed");
  out["R01.foreign-proof-refused"] = await foreignProofs(a.contract);

  world.mode = "completed";
  await a.contract.execute("specimen.change", [], "r1");
  await b.contract.execute("specimen.change", [], "r1");
  const s1 = world.scopes.get("A:r1");
  const sB = world.scopes.get("B:r1");
  if (!s1 || !sB) throw new Error("Scopes were not issued.");
  const attempt = (runId: string, coordination: unknown) => {
    const runtime = approvalRuntime(runId, a.state);
    const result = runtime.revalidateAndConsume(coordination);
    return { admitted: result.consumed, consumed: runtime.consumed.length };
  };
  const carrier: Record<string | symbol, unknown> = {
    runId: s1.runId,
    scope: s1.scope,
    isActive: s1.isActive,
  };
  for (const symbol of Object.getOwnPropertySymbols(s1))
    carrier[symbol] = Reflect.get(s1, symbol);
  let cloneError = "none";
  try {
    structuredClone(s1);
  } catch (error) {
    cloneError = error instanceof Error ? error.name : "unknown";
  }
  // S2: the kernel check alone, without the runtime's own isActive and runId
  // checks.
  const direct = (scope: unknown, runId = "r1") =>
    issuedScope(scope, runId, a.state);
  // S4: issuance itself refuses a run this installation never attached, so no
  // scope exists to present.
  const issuerA = world.issuers.get("A");
  if (!issuerA) throw new Error("Issuer A was not captured.");
  let unattachedIssue: { issued: boolean; namesAttachment: boolean };
  try {
    const scope = issuerA.exclusiveScope("never-attached", "resource-1", () => true);
    unattachedIssue = {
      issued: true,
      namesAttachment: issuedScope(scope, "never-attached", a.state),
    };
  } catch (error) {
    unattachedIssue = {
      issued: false,
      namesAttachment:
        error instanceof Error && /attached run/.test(error.message),
    };
  }
  out["R02.public-index-export"] = {
    exported: typeof issuedScope === "function",
  };
  out["R02.plain-object-refused"] = attempt("r1", {
    runId: "r1",
    scope: "resource-1",
    isActive: () => true,
  });
  out["R02.copy-refused"] = {
    spread: attempt("r1", { ...s1 }),
    symbolCarrier: attempt("r1", carrier),
    cloneError,
  };
  out["R02.other-run-refused"] = {
    ...attempt("r2", s1),
    // S5: the kernel's own run check, without the runtime's runId comparison.
    directOtherRun: direct(s1, "r2"),
    unattachedIssue,
  };
  out["R02.other-installation-refused"] = attempt("r1", sB);
  out["R02.issued-twin-consumed-once"] = {
    ...attempt("r1", s1),
    direct: direct(s1),
  };
  world.scopeMode = "inactive";
  const directInactive = direct(s1);
  const inactive = attempt("r1", s1);
  world.scopeMode = "throw";
  out["R02.inactive-refused"] = {
    ...inactive,
    directInactive,
    directThrowing: direct(s1),
  };
  world.scopeMode = "active";
  await a.contract.execute("specimen.change", [], "r1");
  out["R02.superseded-attachment-refused"] = attempt("r1", s1);
  return out;
}
