// R18 (F01) changing() construction path and the changing-target guard.
// Materialised at starters/contract3/src/addons/state/r18-changing.ts and
// linted there. The test-owned rows arrive as a parameter from the driver,
// because source code never imports tests.
import * as kernel from "../../kernel/index.ts";
import type {
  CauseRow,
  CommandDeclaration,
  DeclaredStation,
  InputSchema,
} from "../../kernel/index.ts";
import {
  admissionInstallation,
  causes,
  coordination,
  input,
  recovery,
  words,
} from "./repair-shared.ts";

type Row = {
  readonly target: string;
  readonly args: readonly string[];
  readonly callerOwned?: string;
};
type KernelValues = Pick<typeof kernel, "command" | "station" | "commonCauses">;
export type R18Spec = {
  readonly rows: Readonly<Record<string, Row>>;
  readonly hold: CauseRow;
  readonly sourceInput: InputSchema;
  readonly stepStation: (
    values: KernelValues,
    source: string,
    arm: "next" | "handoff",
    row: Row,
  ) => DeclaredStation;
  readonly futureBindings: (row: Row) => Readonly<Record<string, string>>;
  readonly targets: (values: KernelValues) => CommandDeclaration[];
  readonly observe: (admit: () => unknown, source: string, row: Row) => unknown;
};

function unchecked(target: unknown, spec: object): unknown {
  if (typeof target !== "function") throw new Error("Not callable.");
  const result: unknown = Reflect.apply(target, undefined, [spec]);
  return result;
}

export function observeR18Changing(spec: R18Spec): Record<string, unknown> {
  const state = admissionInstallation();
  const tableCauses = { ...causes, HOLD: spec.hold };
  const source = (
    identity: string,
    route: readonly string[],
    extra: readonly DeclaredStation[],
    futureBindings: Readonly<Record<string, string>>,
  ) =>
    unchecked(kernel.changing, {
      identity,
      route,
      summary: "Change.",
      input: spec.sourceInput,
      futureBindings,
      capabilities: {},
      effectClass: "repository-local",
      effects: ["e"],
      ifNoResult: state.ifNoResult,
      heldLockCause: "HELD",
      unresolvedCaseCause: "PRIOR",
      stations: [...coordination(identity), ...extra],
      run: () => Promise.reject(new Error("Never entered.")),
    });
  const admit = (commands: readonly unknown[]) => () =>
    unchecked(kernel.contractFor, {
      cli: "specimen",
      commands,
      causes: tableCauses,
      state,
      installedAddons: ["state"],
    });
  const out: Record<string, unknown> = {};
  for (const [suffix, row] of Object.entries(spec.rows))
    for (const arm of ["next", "handoff"] as const) {
      let observation: unknown;
      try {
        const declared = source(
          "specimen.source",
          ["source"],
          [spec.stepStation(kernel, "specimen.source", arm, row)],
          spec.futureBindings(row),
        );
        observation = spec.observe(
          admit([declared, recovery(), ...spec.targets(kernel)]),
          "specimen.source",
          row,
        );
      } catch (error) {
        observation = {
          constructionRefused:
            error instanceof Error ? error.message : String(error),
        };
      }
      out[`R18.${suffix}.${arm}.changing`] = observation;
    }
  const target = source("specimen.target", ["target"], [], {});
  const inspector = kernel.command({
    identity: "specimen.inspector",
    route: ["inspector"],
    summary: "Inspect.",
    input,
    capabilities: {},
    stations: [
      kernel.station(tableCauses, {
        commandIdentity: "specimen.inspector",
        causeCode: "HOLD",
        trigger: "Held.",
        guidance: {
          handoff: {
            owner: "operator",
            reason: "Inspect.",
            inspect: [{ kind: "command", command: "specimen.target", args: [] }],
          },
        },
        repairAction: "Inspect.",
        ...words,
      }),
    ],
    run: () => {
      throw new Error("Never entered.");
    },
  });
  let refused = false;
  try {
    admit([target, inspector, recovery()])();
  } catch {
    refused = true;
  }
  out["R18.u3.changing-target"] = { refused };
  return out;
}
