import { observeAdmissionChanging } from "../../../starters/contract3/src/addons/state/admission-changing.ts";
import { observeContextChanging } from "../../../starters/contract3/src/addons/state/context-changing.ts";
import { observeR18Changing } from "../../../starters/contract3/src/addons/state/r18-changing.ts";
import { observeStateRepair } from "../../../starters/contract3/src/addons/state/state-repair.ts";
import * as r18 from "./r18-spec.ts";

const observers: Readonly<
  Record<string, () => Promise<Record<string, unknown>>>
> = {
  "state-repair": observeStateRepair,
  "admission-changing": () => Promise.resolve(observeAdmissionChanging()),
  "context-changing": observeContextChanging,
  "r18-changing": () => Promise.resolve(observeR18Changing(r18)),
};
const name = process.argv[2] ?? "";
const observer = observers[name];
if (!observer) throw new Error(`Unknown fixture ${name}.`);
process.stdout.write(JSON.stringify(await observer()));
