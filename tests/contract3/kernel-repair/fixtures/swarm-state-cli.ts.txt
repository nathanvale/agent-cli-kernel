// Test-owned process entry for the swarm-triage state cases, materialised at
// tests/contract3/kernel-repair/swarm-state-cli.ts in a disposable tree beside
// a copy of the kernel and the state fixtures, and linted there. argv is
// <scenario> then the CLI argv. Admission scenarios print what contractFor
// did; process scenarios run through the real process front door.
import { appendFileSync } from "node:fs";
import {
  changing,
  contractFor,
  runProcess,
} from "../../../starters/contract3/src/kernel/index.ts";
import {
  authority,
  resource,
} from "../../../starters/contract3/src/kernel/validation/brands.ts";
import {
  causes,
  coordination,
  empty,
  input,
  recovery,
} from "../../../starters/contract3/src/addons/state/repair-shared.ts";
import { approvalAddon } from "./swarm-approval.ts";
import {
  coordinationContract,
  ifNoResultContract,
  ifNoResultTitleContract,
  lateTransactionContract,
  proofAccessorContract,
  recordingInstallation,
  runReuse,
  spreadChangingContract,
} from "../../../starters/contract3/src/addons/state/swarm-state.ts";

function admission(compose: () => object): void {
  try {
    compose();
    process.stdout.write(JSON.stringify({ admission: "admitted" }));
  } catch (error) {
    process.stdout.write(
      JSON.stringify({
        admission: "refused",
        reason: error instanceof Error ? error.message : String(error),
      }),
    );
  }
}
const ledger = process.env.LEDGER ?? "";
const note = (line: string) => appendFileSync(ledger, `${line}\n`);
// X-S6: a changing command behind approval whose state installation and
// handler both report to the ledger.
function guardedContract() {
  const state = recordingInstallation(() => note("state-attached"));
  const guarded = changing({
    identity: "specimen.guarded",
    route: ["guarded"],
    summary: "Apply e behind approval.",
    input: { ...input, deadlineSupported: true },
    capabilities: { approval: true },
    approval: () => ({
      authority: authority("specimen.owner"),
      resource: resource("specimen.item"),
      resourceRevision: "1",
      effects: ["e"],
      preview: "preview-1",
    }),
    effectClass: "repository-local",
    effects: ["e"],
    ifNoResult: state.ifNoResult,
    heldLockCause: "HELD",
    unresolvedCaseCause: "PRIOR",
    stations: coordination("specimen.guarded"),
    run: (_input, context) => {
      note("handler-entered");
      return Promise.resolve(
        context.result(
          { commandIdentity: "specimen.guarded", causeCode: "HELD" },
          { message: "Held.", data: null, effects: empty },
        ),
      );
    },
  });
  return contractFor({
    cli: "specimen",
    commands: [guarded, recovery()],
    causes,
    state,
    installedAddons: ["state", "approval"],
  });
}
const scenarios: Readonly<Record<string, (argv: string[]) => Promise<unknown>>> =
  {
    "ifnoresult-required": () =>
      Promise.resolve(admission(() => ifNoResultContract(true))),
    "ifnoresult-none": (argv) => runProcess(ifNoResultContract(false))(argv),
    "ifnoresult-title-admission": () =>
      Promise.resolve(
        admission(() => ifNoResultTitleContract(process.env.VALUE ?? "")),
      ),
    "ifnoresult-title": (argv) =>
      runProcess(ifNoResultTitleContract(process.env.VALUE ?? ""))(argv),
    "coordination-lock-common": () =>
      Promise.resolve(admission(() => coordinationContract("lock-common"))),
    "coordination-unresolved-common": () =>
      Promise.resolve(
        admission(() => coordinationContract("unresolved-common")),
      ),
    "spread-changing-admission": () =>
      Promise.resolve(admission(() => spreadChangingContract())),
    "coordination-host": () =>
      Promise.resolve(admission(() => coordinationContract("host"))),
    "proof-accessor": (argv) => runProcess(proofAccessorContract(false))(argv),
    "proof-steady": (argv) => runProcess(proofAccessorContract(true))(argv),
    "late-transaction": (argv) =>
      runProcess(
        lateTransactionContract(Number(process.env.HOPS ?? "0"), (line) =>
          note(line),
        ),
      )(argv),
    "approval-attach": (argv) =>
      runProcess(guardedContract(), {
        addons: { approval: approvalAddon("specimen.guarded", note) },
      })(argv),
    // F2: a denied check phase claims nothing, so the same run identity is
    // then granted and entered.
    "deny-then-grant": async () => {
      const contract = guardedContract();
      const attempt = async (granted: boolean) => {
        const executed = await contract.execute("specimen.guarded", [], "R", {
          approval: { check: () => Promise.resolve(granted) },
        });
        return executed.ok ? "entered" : executed.path;
      };
      process.stdout.write(
        JSON.stringify({ denied: await attempt(false), granted: await attempt(true) }),
      );
    },
    "run-reuse": async () =>
      process.stdout.write(
        JSON.stringify(await runReuse(process.env.SECOND_RUN_ID ?? "")),
      ),
  };
const [scenario = "", ...argv] = process.argv.slice(2);
const run = scenarios[scenario];
if (!run) throw new Error(`Unknown scenario ${scenario}.`);
await run(argv);
