// R12 (F10) changing-command arms. Prints observations keyed by frozen case
// identity; the parent test owns the expected values.
import { changing, command, commonCauses, contractFor, station } from "../../kernel/index.ts";
const input = { arguments: [], options: [], constraints: [], deadlineSupported: false, additionalArguments: false, additionalOptions: false } as const;
const empty = { completed: [], remaining: [], uncertain: [], inventoryComplete: true } as const;
const refusal = { outcome: "refused", transactionState: "unchanged", failureClass: "domain", exitCode: 3, retryable: false, guidance: "handoff" } as const;
const causes = { HELD: refusal, PRIOR: refusal } as const;
const words = { retryDelayPolicy: { kind: "none" }, reachability: "required", unreachableRationale: null, independentObservable: "context-fixture" } as const;
const state = contractFor.stateAddon((issuer) => ({
  ifNoResult: issuer.ifNoResult({ kind: "command", command: "specimen.pending", args: [] }),
  attach(runId) {
    return { async transaction() { return issuer.transactionProof(runId, { transactionState: "unchanged", effects: empty }); } };
  },
}));
const recovery = command({ identity: "specimen.pending", route: ["pending"], summary: "Inspect pending.", input, capabilities: {}, stations: [station(commonCauses, { commandIdentity: "specimen.pending", causeCode: "SUCCESS_UNCHANGED", trigger: "Read pending.", guidance: { next: { kind: "instruction", text: "Inspect pending state." } }, repairAction: null, ...words })], run: (_input, context) => context.result({ commandIdentity: "specimen.pending", causeCode: "SUCCESS_UNCHANGED" }, { message: "Read pending.", data: null, effects: empty }) });
const coordination = (["HELD", "PRIOR"] as const).map((causeCode) => station(causes, { commandIdentity: "specimen.change", causeCode, trigger: "Coordination refusal.", repairAction: "Inspect.", guidance: { handoff: { owner: "operator", reason: "Inspect.", inspect: [{ kind: "instruction", text: "Inspect." }] } }, ...words }));
const unchanged = station(commonCauses, { commandIdentity: "specimen.change", causeCode: "SUCCESS_UNCHANGED", trigger: "Unchanged.", guidance: { next: { kind: "done" } }, repairAction: null, ...words });
const received: unknown[] = [];
const change = changing({ identity: "specimen.change", route: ["change"], summary: "Change.", input, capabilities: { diagnostics: true }, effectClass: "repository-local", effects: ["e"], ifNoResult: state.ifNoResult, heldLockCause: "HELD", unresolvedCaseCause: "PRIOR", stations: [...coordination, unchanged], run: async (_input, context) => {
  received.push(context.diagnostics);
  return context.result({ commandIdentity: "specimen.change", causeCode: "SUCCESS_UNCHANGED" }, { message: "Unchanged.", data: null, transaction: await context.state.transaction(async () => {}) });
} });
const contract = contractFor({ cli: "specimen", commands: [change, recovery], causes, state, installedAddons: ["state", "diagnostics"] });
const diagnostics = () => ({ record() {} });
async function observe(context: object, validated: () => unknown) {
  received.length = 0;
  const executed = await contract.execute("specimen.change", [], `run-${Math.random()}`, context);
  return { executed: executed.ok, entries: received.length, sameAsValidated: received.length === 1 && received[0] === validated() };
}
const out: Record<string, unknown> = {};
{
  const capability = diagnostics();
  const context = {};
  Object.defineProperty(context, "diagnostics", { value: capability, enumerable: false });
  out["R12.changing-hidden-capability"] = await observe(context, () => capability);
}
{
  const first = diagnostics();
  let reads = 0;
  const context = {};
  Object.defineProperty(context, "diagnostics", { enumerable: true, get() { reads++; return reads === 1 ? first : {}; } });
  out["R12.changing-getter-capability"] = await observe(context, () => first);
}
{
  const capability = diagnostics();
  out["R12.changing-plain-twin"] = await observe({ diagnostics: capability }, () => capability);
}
process.stdout.write(JSON.stringify(out));
