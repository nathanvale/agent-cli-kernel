// R12 (F10) changing-command arms. Materialised at
// starters/contract3/src/addons/state/context-changing.ts and linted there.
import {
  changing,
  commonCauses,
  contractFor,
  station,
} from "../../kernel/index.ts";
import {
  causes,
  coordination,
  empty,
  input,
  recovery,
  words,
} from "./repair-shared.ts";

function fixture() {
  const received: unknown[] = [];
  const state = contractFor.stateAddon((issuer) => ({
    ifNoResult: issuer.ifNoResult({
      kind: "command",
      command: "specimen.pending",
      args: [],
    }),
    attach(runId) {
      return {
        transaction: () =>
          Promise.resolve(
            issuer.transactionProof(runId, {
              transactionState: "unchanged",
              effects: empty,
            }),
          ),
      };
    },
  }));
  const change = changing({
    identity: "specimen.change",
    route: ["change"],
    summary: "Change.",
    input,
    capabilities: { diagnostics: true },
    effectClass: "repository-local",
    effects: ["e"],
    ifNoResult: state.ifNoResult,
    heldLockCause: "HELD",
    unresolvedCaseCause: "PRIOR",
    stations: [
      ...coordination("specimen.change"),
      station(commonCauses, {
        commandIdentity: "specimen.change",
        causeCode: "SUCCESS_UNCHANGED",
        trigger: "Unchanged.",
        guidance: { next: { kind: "done" } },
        repairAction: null,
        ...words,
      }),
    ],
    run: async (_input, context) => {
      received.push(context.diagnostics);
      return context.result(
        { commandIdentity: "specimen.change", causeCode: "SUCCESS_UNCHANGED" },
        {
          message: "Unchanged.",
          data: null,
          transaction: await context.state.transaction(() => Promise.resolve()),
        },
      );
    },
  });
  const contract = contractFor({
    cli: "specimen",
    commands: [change, recovery()],
    causes,
    state,
    installedAddons: ["state", "diagnostics"],
  });
  return { contract, received };
}
const diagnostics = () => ({
  record() {
    return undefined;
  },
});

export async function observeContextChanging(): Promise<
  Record<string, unknown>
> {
  const { contract, received } = fixture();
  let runs = 0;
  const observe = async (context: object, validated: unknown) => {
    received.length = 0;
    const executed = await contract.execute(
      "specimen.change",
      [],
      `run-${++runs}`,
      context,
    );
    return {
      executed: executed.ok,
      entries: received.length,
      sameAsValidated: received.length === 1 && received[0] === validated,
    };
  };
  const hidden = diagnostics();
  const hiddenContext = {};
  Object.defineProperty(hiddenContext, "diagnostics", {
    value: hidden,
    enumerable: false,
  });
  const first = diagnostics();
  let reads = 0;
  const getterContext = {};
  Object.defineProperty(getterContext, "diagnostics", {
    enumerable: true,
    get() {
      reads++;
      return reads === 1 ? first : {};
    },
  });
  const plain = diagnostics();
  return {
    "R12.changing-hidden-capability": await observe(hiddenContext, hidden),
    "R12.changing-getter-capability": await observe(getterContext, first),
    "R12.changing-plain-twin": await observe({ diagnostics: plain }, plain),
  };
}
