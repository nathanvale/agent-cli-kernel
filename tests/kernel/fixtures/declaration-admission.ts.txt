import { expect } from "bun:test";
import { changing, contractFor, station } from "../../kernel/index.ts";
const input = { arguments: [], options: [], constraints: [], deadlineSupported: false, additionalArguments: false, additionalOptions: false } as const;
const row = { outcome: "refused", transactionState: "unchanged", failureClass: "domain", exitCode: 3, retryable: false, guidance: "handoff" } as const;
const causes = { DOMAIN_HELD: row, DOMAIN_PRIOR: row } as const;
const state = contractFor.stateAddon((issuer) => ({ ifNoResult: issuer.ifNoResult({kind: "command", command: "specimen.pending", args: []}), attach() { throw new Error("No state execution in declaration fixture."); } }));
const flags = { ...input, options: [{ name: "--change", aliases: [], summary: "Select change.", required: false, repeatable: false, value: null }] } as const;
function declare(identity: string, root = false) {
  const stations = (["DOMAIN_HELD", "DOMAIN_PRIOR"] as const).map((causeCode) => station(causes, {commandIdentity: identity, causeCode, trigger: "Coordination refusal.", repairAction: "Inspect.", guidance: {handoff: {owner: "operator", reason: "Inspect.", inspect: [{kind: "instruction", text: "Inspect."}]}}, retryDelayPolicy: {kind: "none"}, reachability: "required", unreachableRationale: null, independentObservable: "admission-only"}));
  return changing({identity, route: root ? [] : ["change"], summary: "Change.", input: root ? flags : input, capabilities: {}, effectClass: "repository-local", effects: ["write"], ifNoResult: state.ifNoResult, heldLockCause: "DOMAIN_HELD", unresolvedCaseCause: "DOMAIN_PRIOR", stations, run: async () => { throw new Error("Declaration fixture never enters handler."); }});
}
const tally = { validIdentities: 0, oversizedRejected: 0, rootInvocationControls: 0 };
for (const identity of ["a".repeat(128), "é".repeat(64)]) {
  expect(declare(identity).commandIdentity).toBe(identity);
  tally.validIdentities++;
  expect(() => declare(`${identity}a`)).toThrow("128 UTF-8 bytes");
  tally.oversizedRejected++;
}
const root = declare("specimen.change", true);
expect(await root.execute(["--"], {}, "empty-root")).toMatchObject({ok: false, path: "input"});
tally.rootInvocationControls++;
expect(await root.execute(["--change"], {}, "supplied-root")).toMatchObject({ok: false, path: "capabilities"});
tally.rootInvocationControls++;
process.stdout.write(JSON.stringify(tally));
