import {
  changing,
  command,
  commonCauses,
  contractFor,
  station,
} from "../../kernel/index.ts";
import type {
  EffectTransaction,
  StateAddonDescriptor,
  StateObservation,
} from "../../kernel/index.ts";
import { causes, coordination, empty, input, recovery, words } from "./repair-shared.ts";

type Observe = <Effect extends string>(
  effects: readonly Effect[],
) => StateObservation<Effect>;

const completedObservation: Observe = <Effect extends string>(
  effects: readonly Effect[],
) => {
  const [first, ...rest] = effects;
  if (first === undefined) throw new Error("Fixture requires an effect.");
  const observation: StateObservation<Effect> = {
    transactionState: "completed",
    effects: {
      completed: [first, ...rest],
      remaining: [],
      uncertain: [],
      inventoryComplete: true,
    },
  };
  return observation;
};

function installation(
  observe: Observe,
  pendingArgs: readonly string[] = [],
  onAttach: () => void = () => undefined,
): StateAddonDescriptor {
  return contractFor.stateAddon((issuer) => ({
    ifNoResult: issuer.ifNoResult({
      kind: "command",
      command: "specimen.pending",
      args: [...pendingArgs],
    }),
    attach<Effect extends string>(runId: string, effects: readonly Effect[]) {
      onAttach();
      return {
        async transaction(
          work: (transaction: EffectTransaction<Effect>) => Promise<void>,
        ) {
          await work({
            record: (effect, operation) =>
              operation(
                issuer.recordedEffect(runId, effect, (_run, _effect, start) =>
                  start(),
                ),
              ),
          });
          return issuer.transactionProof(runId, observe(effects));
        },
      };
    },
  }));
}

const held = { commandIdentity: "specimen.apply", causeCode: "DOMAIN_HELD" } as const;
const heldFacts = { message: "Held.", data: null, effects: empty } as const;

type Handler =
  | { readonly kind: "throw" }
  | { readonly kind: "transaction-then-held" }
  | {
      readonly kind: "late-transaction";
      readonly hops: number;
      readonly record: (line: string) => void;
    };

function applyContract(
  state: StateAddonDescriptor,
  handler: Handler,
  pending: ReturnType<typeof recovery> | ReturnType<typeof command> = recovery(),
  spread = false,
) {
  const apply = changing({
    identity: "specimen.apply",
    route: ["apply"],
    summary: "Apply e.",
    input,
    capabilities: {},
    effectClass: "repository-local",
    effects: ["e"],
    ifNoResult: state.ifNoResult,
    heldLockCause: "DOMAIN_HELD",
    unresolvedCaseCause: "DOMAIN_PRIOR",
    stations: coordination("specimen.apply"),
    run: async (_input, context) => {
      if (handler.kind === "throw") throw new Error("Handler fault.");
      if (handler.kind === "transaction-then-held") {
        await context.state.transaction(() => Promise.resolve());
        return context.result(held, heldFacts);
      }
      const { hops, record } = handler;
      let chain = Promise.resolve();
      for (let hop = 0; hop < hops; hop++) chain = chain.then(() => undefined);
      void chain
        .then(() =>
          context.state.transaction(() => {
            record("effect-applied");
            return Promise.resolve();
          }),
        )
        .then(
          () => record("proof-issued"),
          (error: unknown) =>
            record(
              `transaction-refused: ${error instanceof Error ? error.message : String(error)}`,
            ),
        );
      record("handler-returned");
      return context.result(held, heldFacts);
    },
  });
  return contractFor({
    cli: "specimen",
    commands: [spread ? { ...apply } : apply, pending],
    causes,
    state,
    installedAddons: ["state"],
  });
}

export function ifNoResultContract(requiredOperand: boolean) {
  const pending = command({
    identity: "specimen.pending",
    route: ["pending"],
    summary: "Inspect pending.",
    input: requiredOperand
      ? {
          ...input,
          arguments: [
            {
              name: "scope",
              summary: "Scope.",
              required: true,
              value: {
                kind: "string",
                minLength: 1,
                maxLength: 20,
                choices: null,
                pattern: null,
              },
            },
          ],
        }
      : input,
    capabilities: {},
    stations: [
      station(commonCauses, {
        commandIdentity: "specimen.pending",
        causeCode: "SUCCESS_UNCHANGED",
        trigger: "Read pending.",
        guidance: {
          next: { kind: "instruction", text: "Inspect pending state." },
        },
        repairAction: null,
        ...words,
      }),
    ],
    run: (_input, context) =>
      context.result(
        { commandIdentity: "specimen.pending", causeCode: "SUCCESS_UNCHANGED" },
        { message: "Read pending.", data: null, effects: empty },
      ),
  });
  return applyContract(
    installation(completedObservation),
    { kind: "throw" },
    pending,
  );
}

        trigger: "Read pending.",
        guidance: {
          next: { kind: "instruction", text: "Inspect pending state." },

      context.result(

    installation(completedObservation, ["--title", value]),
    { kind: "throw" },
    pending,
  );
}

export function coordinationContract(
  variant: "lock-common" | "unresolved-common" | "host",
) {
  const state = installation(completedObservation);
  const table = { ...commonCauses, ...causes };
  const heldCause =
    variant === "lock-common" ? "DOMAIN_AUTHORITY_REQUIRED" : "DOMAIN_HELD";
  const priorCause =
    variant === "unresolved-common" ? "DOMAIN_AUTHORITY_REQUIRED" : "DOMAIN_PRIOR";
  const refusal = {
    trigger: "Coordination refusal.",
    repairAction: "Inspect the lock.",
    guidance: {
      handoff: {
        owner: "operator",
        reason: "Inspect the lock owner.",
        inspect: [{ kind: "instruction", text: "Inspect the lock." }],
      },
    },
    ...words,
  } as const;
  const apply = changing({
    identity: "specimen.apply",
    route: ["apply"],
    summary: "Apply e.",
    input,
    capabilities: {},
    effectClass: "repository-local",
    effects: ["e"],
    ifNoResult: state.ifNoResult,
    heldLockCause: heldCause,
    unresolvedCaseCause: priorCause,
    stations: [
      station(table, {
        ...refusal,
        commandIdentity: "specimen.apply",
        causeCode: heldCause,
      }),
      station(table, {
        ...refusal,
        commandIdentity: "specimen.apply",
        causeCode: priorCause,
      }),
    ],
    run: () => Promise.reject(new Error("Admission fixture never runs.")),
  });
  return contractFor({
    cli: "specimen",
    commands: [apply, recovery()],
    causes,
    state,
    installedAddons: ["state"],
  });
}

export function proofAccessorContract(steady: boolean) {
  const accessor: Observe = <Effect extends string>(
    effects: readonly Effect[],
  ) => {
    if (steady) {
      const unchanged: StateObservation<Effect> = {
        transactionState: "unchanged",
        effects: empty,
      };
      return unchanged;
    }
    const observation = completedObservation(effects);
    let reads = 0;
    Object.defineProperty(observation, "transactionState", {
      enumerable: true,
      get: () => (reads++ === 0 ? "completed" : "unchanged"),
    });
    return observation;
  };
  return applyContract(installation(accessor), {
    kind: "transaction-then-held",
  });
}

export function lateTransactionContract(
  hops: number,
  record: (line: string) => void,
) {
  return applyContract(installation(completedObservation), {
    kind: "late-transaction",
    hops,
    record,
  });
}

export async function runReuse(secondRunId: string) {
  let calls = 0;
  const state = installation(completedObservation);
  const done = station(commonCauses, {
    commandIdentity: "specimen.apply",
    causeCode: "SUCCESS_COMPLETED",
    trigger: "Applied.",
    guidance: { next: { kind: "done" } },
    repairAction: null,
    ...words,
  });
  const apply = changing({
    identity: "specimen.apply",
    route: ["apply"],
    summary: "Apply e.",
    input,
    capabilities: {},
    effectClass: "repository-local",
    effects: ["e"],
    ifNoResult: state.ifNoResult,
    heldLockCause: "DOMAIN_HELD",
    unresolvedCaseCause: "DOMAIN_PRIOR",
    stations: [...coordination("specimen.apply"), done],
    run: async (_input, context) => {
      calls++;
      if (calls > 1) return context.result(held, heldFacts);
      return context.result(
        { commandIdentity: "specimen.apply", causeCode: "SUCCESS_COMPLETED" },
        {
          message: "Applied.",
          data: null,
          transaction: await context.state.transaction(() => Promise.resolve()),
        },
      );
    },
  });
  const contract = contractFor({
    cli: "specimen",
    commands: [apply, recovery()],
    causes,
    state,
    installedAddons: ["state"],
  });
  const view = async (runId: string) => {
    const executed = await contract.execute("specimen.apply", [], runId);
    if (!executed.ok) return { executed: false, path: executed.path };
    const built = await contract.buildResult(executed.value);
    return built.ok
      ? {
          executed: true,
          runId: built.value.result.runId,
          causeCode: built.value.result.causeCode,
          transactionState: built.value.result.transactionState,
          completed: built.value.result.effects.completed,
        }
      : { executed: true, built: false };
  };
  return {
    first: await view("R"),
    second: await view(secondRunId),
    third: await view("R"),
  };
}

export function recordingInstallation(onAttach: () => void) {
  return installation(completedObservation, [], onAttach);
}

export function spreadChangingContract() {
  return applyContract(
    installation(completedObservation),
    { kind: "throw" },
    recovery(),
    true,
  );
}
