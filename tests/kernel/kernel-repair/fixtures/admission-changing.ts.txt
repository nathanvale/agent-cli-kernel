import { changing, commonCauses, contractFor } from "../../kernel/index.ts";
import {
  admissionInstallation,
  causes,
  coordination,
  input,
  recovery,
  words,
} from "./repair-shared.ts";

const flags = {
  ...input,
  options: [
    {
      name: "--change",
      aliases: [],
      summary: "Select change.",
      required: false,
      repeatable: false,
      value: null,
    },
  ],
} as const;
const reserved = [
  "INTERNAL_PREPARATION",
  "INTERNAL_RESULT_UNCHANGED",
  "INTERNAL_RESULT_COMPLETED",
  "INTERNAL_RESULT_PARTIAL",
  "INTERNAL_RESULT_UNKNOWN",
] as const;
const tokens = [
  "",
  " ",
  "--json",
  "--deadline-ms",
  "--help",
  "--discover",
  "--discover-command",
  "--",
  "-x",
];

function unchecked(target: unknown, spec: object): unknown {
  if (typeof target !== "function") throw new Error("Not callable.");
  const result: unknown = Reflect.apply(target, undefined, [spec]);
  return result;
}
function outcome(action: () => unknown) {
  try {
    action();
    return { refused: false, message: "" };
  } catch (error) {
    return {
      refused: true,
      message: error instanceof Error ? error.message : String(error),
    };
  }
}
function literal(causeCode: (typeof reserved)[number]) {
  const row = commonCauses[causeCode];
  return {
    commandIdentity: "specimen.change",
    causeCode,
    row,
    trigger: "HOST-WORDS trigger.",
    repairAction: "HOST-WORDS repair.",
    guidance:
      row.guidance === "handoff"
        ? {
            handoff: {
              owner: "operator",
              reason: "HOST-WORDS reason.",
              inspect: [{ kind: "instruction", text: "HOST-WORDS inspect." }],
            },
          }
        : { next: { kind: "instruction", text: "HOST-WORDS next." } },
    ...words,
  };
}

export function observeAdmissionChanging(): Record<string, unknown> {
  const state = admissionInstallation();
  const handled = { entered: 0 };
  const spec = (route: readonly string[], extra: readonly object[]) => ({
    identity: "specimen.change",
    route,
    summary: "Change.",
    input: flags,
    capabilities: {},
    effectClass: "repository-local",
    effects: ["a", "b"],
    ifNoResult: state.ifNoResult,
    heldLockCause: "HELD",
    unresolvedCaseCause: "PRIOR",
    stations: [...coordination("specimen.change"), ...extra],
    run: () => {
      handled.entered++;
      return Promise.reject(new Error("Admission fixture never enters a handler."));
    },
  });
  const reservedObservations = reserved.map((cause) => {
    const result = outcome(() =>
      unchecked(contractFor, {
        cli: "specimen",
        commands: [
          unchecked(changing, spec(["change"], [literal(cause)])),
          recovery(),
        ],
        causes,
        state,
        installedAddons: ["state"],
      }),
    );
    return {
      cause,
      refused: result.refused,
      reserved: /reserved/i.test(result.message),
    };
  });
  return {
    "R03.changing-literal-refused": reservedObservations,
    "R04.changing-builder-refused": tokens.map((token) => ({
      token,
      refused: outcome(() => unchecked(changing, spec([token], []))).refused,
    })),
    entered: handled.entered,
  };
}
