type Fields = ReadonlyMap<string, unknown>;
function fields(value: unknown): Fields {
  if (typeof value !== "object" || value === null || Array.isArray(value))
    throw new Error("Driver input must be an object.");
  return new Map<string, unknown>(Object.entries(value));
}
function text(map: Fields, key: string): string {
  const value = map.get(key);
  if (typeof value !== "string") throw new Error(`Driver ${key} must be text.`);
  return value;
}
function texts(map: Fields, key: string): readonly string[] {
  const value = map.get(key);
  if (!Array.isArray(value)) throw new Error(`Driver ${key} must be a list.`);
  const items: unknown[] = value;
  return items.map((item) => {
    if (typeof item !== "string") throw new Error(`Driver ${key} item.`);
    return item;
  });
}
function oneOf<const T extends string>(
  map: Fields,
  key: string,
  options: readonly T[],
): T {
  const value = map.get(key);
  const found = options.find((option) => option === value);
  if (found === undefined) throw new Error(`Driver ${key} is not admitted.`);
  return found;
}
function flag(map: Fields, key: string): boolean {
  const value = map.get(key) ?? false;
  if (typeof value !== "boolean") throw new Error(`Driver ${key} flag.`);
  return value;
}

const barrierNames = [
  "after-preflight",
  "after-acquire",
  "after-consume",
  "after-effect",
] as const;
type Barrier = (typeof barrierNames)[number];
type World = {
  readonly grants: string;
  readonly consumed: string;
  readonly effects: string;
  readonly revision: string;
  readonly clock: string;
  readonly lock: string;
  readonly contacts: string;
};
type Common = {
  readonly world: World;
  readonly runId: string;
  readonly lock: "refuse" | "wait" | "private";
  readonly coordination:
    | "held"
    | "released"
    | "foreign-scope"
    | "foreign-run"
    | "lost-after-consume"
    | "forged";
  readonly lostMarker: string;
  readonly barriers: readonly Barrier[];
};

function world(map: Fields): World {
  const value = fields(map.get("world"));
  return {
    grants: text(value, "grants"),
    consumed: text(value, "consumed"),
    effects: text(value, "effects"),
    revision: text(value, "revision"),
    clock: text(value, "clock"),
    lock: text(value, "lock"),
    contacts: text(value, "contacts"),
  };
}
function common(map: Fields): Common {
  const barriers = texts(map, "barriers").map((name) => {
    const found = barrierNames.find((item) => item === name);
    if (found === undefined) throw new Error("Unknown barrier.");
    return found;
  });
  return {
    world: world(map),
    runId: text(map, "runId"),
    lock: oneOf(map, "lock", ["refuse", "wait", "private"]),
    coordination: map.has("coordination")
      ? oneOf(map, "coordination", [
          "held",
          "released",
          "foreign-scope",
          "foreign-run",
          "lost-after-consume",
          "forged",
        ])
      : "held",
    lostMarker: map.has("lostMarker") ? text(map, "lostMarker") : "",
    barriers,
  };
}

const raw: unknown = JSON.parse(process.argv[2] ?? "null");
const input = fields(raw);
const lines = createInterface({ input: process.stdin })[
  Symbol.asyncIterator
]();
function emit(event: Record<string, unknown>): void {
  process.stdout.write(`${JSON.stringify(event)}\n`);
}
function failure(error: unknown): Record<string, unknown> {
  return error instanceof ApprovalRejection
    ? { error: error.reason, consumed: error.consumed }
    : { error: "unexpected" };
}
async function barrier(
  name: Barrier,
  wanted: readonly Barrier[],
): Promise<void> {
  if (!wanted.includes(name)) return;
  emit({ barrier: name });
  const line = await lines.next();
  if (line.done === true) throw new Error("Parent closed the barrier.");
}
function now(place: World): number {
  return Number(readFileSync(place.clock, "utf8"));
}
function revision(place: World): string {
  return readFileSync(place.revision, "utf8");
}

type Forged = {
  readonly runId: string;
  readonly scope: string;
  readonly isActive: () => boolean;
};
type Held =
  | { readonly lock: FixtureLock; readonly scope: ExclusiveScope }
  | { readonly lock: FixtureLock; readonly forged: Forged };

async function acquire(
  settings: Common,
  scopeName: string,
): Promise<Held | undefined> {
  const lock = new FixtureLock(
    settings.lock === "private"
      ? `${settings.world.lock}-${settings.runId}`
      : settings.world.lock,
  );
  const acquired =
    settings.lock === "wait"
      ? await lock.waitAcquire(10_000)
      : lock.tryAcquire();
  if (!acquired) return undefined;
  const active = () => {
    if (
      settings.coordination === "lost-after-consume" &&
      existsSync(settings.lostMarker)
    )
      lock.release();
    return lock.active();
  };
  if (settings.coordination === "forged")
    return {
      lock,
      forged: { runId: settings.runId, scope: scopeName, isActive: active },
    };
  const scope = exclusiveScope(
    settings.coordination === "foreign-run"
      ? `${settings.runId}-other`
      : settings.runId,
    settings.coordination === "foreign-scope" ? `${scopeName}-other` : scopeName,
    active,
  );
  if (settings.coordination === "released") lock.release();
  return { lock, scope };
}

const boundary = contractFor({ cli: "boundary", commands: [] }).boundary;

async function untyped(
  method: (...args: never[]) => unknown,
  owner: object,
  args: readonly unknown[],
): Promise<Fields> {
  const result: unknown = await Reflect.apply(method, owner, args);
  return fields(result);
}

function failCloseAfter(marker: string): void {
  const original = { ...nodeFs };
  let armed = true;
  const wrapped: Record<string, unknown> = {
    ...original,
    closeSync(descriptor: number): void {
      original.closeSync(descriptor);
      if (armed && original.existsSync(marker)) {
        armed = false;
        const failure = new Error("Injected close failure.");
        Reflect.set(failure, "code", "EIO");
        throw failure;
      }
    },
  };
  wrapped.default = wrapped;
  void mock.module("node:fs", () => wrapped);
}

async function apply(map: Fields): Promise<void> {
  const settings = common(map);
  const place = settings.world;
  const preview = text(map, "preview");
  const dropPreview = flag(map, "dropPreviewUnderLock");
  const closeFailure = map.has("failCloseAfter")
    ? text(map, "failCloseAfter")
    : "";
  const runtime = approval({
    operation: text(map, "operation"),
    runId: settings.runId,
    store: { grants: place.grants, consumed: place.consumed },
    now: () => now(place),
  }).runtime(state);
  const request = (resourceRevision: string, withPreview: boolean) => ({
    authority: boundary.authority(text(map, "authority")),
    resource: boundary.resource(text(map, "resource")),
    resourceRevision,
    effects: texts(map, "effects"),
    ...(withPreview ? { preview } : {}),
  });
  let admitted: boolean;
  try {
    admitted = await runtime.check(request(revision(place), true));
  } catch (error) {
    return emit({ phase: "preflight", ...failure(error) });
  }
  emit({ phase: "preflight", admitted });
  if (!admitted) return;
  await barrier("after-preflight", settings.barriers);
  const held = await acquire(settings, text(map, "resource"));
  if (!held) return emit({ phase: "lock-refused" });
  emit({ phase: "acquired" });
  await barrier("after-acquire", settings.barriers);
  if (closeFailure) failCloseAfter(closeFailure);
  let consumed: boolean;
  try {
    const current = request(revision(place), !dropPreview);
    if ("forged" in held) {
      const admitted = (
        await untyped(runtime.revalidateAndConsume, runtime, [
          current,
          held.forged,
        ])
      ).get("consumed");
      if (typeof admitted !== "boolean") throw new Error("Admission shape.");
      consumed = admitted;
    } else
      consumed = (await runtime.revalidateAndConsume(current, held.scope))
        .consumed;
  } catch (error) {
    held.lock.release();
    return emit({ phase: "consume", ...failure(error) });
  }
  emit({ phase: "consume", admitted: consumed });
  if (!consumed) return held.lock.release();
  await barrier("after-consume", settings.barriers);
  const descriptor = openSync(place.effects, "a");
  writeSync(descriptor, `${settings.runId}\n`);
  fsyncSync(descriptor);
  closeSync(descriptor);
  emit({ phase: "effect" });
  await barrier("after-effect", settings.barriers);
  held.lock.release();
  emit({ phase: "done" });
}

function closureItem(map: Fields): ClosureItem {
  return {
    operation: text(map, "operation"),
    disposition: oneOf(map, "disposition", ["outcome-unknown"]),
    evidence: text(map, "evidence"),
    scope: text(map, "scope"),
  };
}
function stopItem(map: Fields): StopItem {
  const liveConflict = map.get("liveConflict");
  if (typeof liveConflict !== "boolean") throw new Error("liveConflict.");
  return {
    operation: text(map, "operation"),
    owner: text(map, "owner"),
    writers: texts(map, "writers"),
    scope: text(map, "scope"),
    liveConflict,
  };
}
function readItem(path: string): Fields {
  const parsed: unknown = JSON.parse(readFileSync(path, "utf8"));
  return fields(parsed);
}

async function authority(map: Fields): Promise<void> {
  const settings = common(map);
  const place = settings.world;
  const kind = oneOf(map, "kind", ["closure", "stop"]);
  const itemPath = text(map, "item");
  const source = adversarialSource(text(map, "source"));
  const verifier = authorityVerifier({
    runId: settings.runId,
    consumed: place.consumed,
    now: () => now(place),
    ...(kind === "closure" ? { closure: source } : { stop: source }),
  });
  type Scope = Parameters<typeof verifier.closure.consume>[1];
  const step = (scope?: Scope) => {
    const fresh = readItem(itemPath);
    if (kind === "closure") {
      const item = closureItem(fresh);
      return scope
        ? verifier.closure.consume(item, scope, state)
        : verifier.closure.preflight(item);
    }
    const item = stopItem(fresh);
    return scope
      ? verifier.stop.consume(item, scope, state)
      : verifier.stop.preflight(item);
  };
  const forgedStep = async (forged: Forged) => {
    const fresh = readItem(itemPath);
    const [slot, item] =
      kind === "closure"
        ? [verifier.closure, closureItem(fresh)]
        : [verifier.stop, stopItem(fresh)];
    return text(
      await untyped(slot.consume, slot, [item, forged, state]),
      "status",
    );
  };
  const first = await step();
  emit({ phase: "preflight", status: first.status });
  if (first.status !== "verified") return;
  await barrier("after-preflight", settings.barriers);
  const held = await acquire(settings, text(readItem(itemPath), "scope"));
  if (!held) return emit({ phase: "lock-refused" });
  emit({ phase: "acquired" });
  await barrier("after-acquire", settings.barriers);
  try {
    const verdict =
      "forged" in held
        ? await forgedStep(held.forged)
        : (await step(held.scope)).status;
    emit({ phase: "consume", status: verdict });
  } catch (error) {
    emit({ phase: "consume", ...failure(error) });
  }
  await barrier("after-consume", settings.barriers);
  held.lock.release();
  emit({ phase: "done" });
}

async function hold(place: World): Promise<void> {
  const lock = new FixtureLock(place.lock);
  if (!lock.tryAcquire()) return emit({ phase: "lock-refused" });
  emit({ barrier: "holding" });
  await lines.next();
  lock.release();
  emit({ phase: "done" });
}

const handoff = {
  owner: "operator",
  reason: "Fixture coordination refusal.",
  inspect: [
    { kind: "command", command: "fixture.case", args: ["{binding.preview}"] },
  ],
} as const;
const refusal = {
  outcome: "refused",
  transactionState: "unchanged",
  failureClass: "domain",
  exitCode: 3,
  retryable: false,
  guidance: "handoff",
} as const;
const attachmentHandoff = {
  owner: "operator",
  reason: "Fixture attachment refusal.",
  inspect: [{ kind: "instruction", text: "Inspect the fixture run." }],
} as const;
const causes = { DOMAIN_FIXTURE_LOCK_HELD: refusal, DOMAIN_FIXTURE_UNRESOLVED: refusal };
function stationFor<
  const Identity extends "fixture.approve" | "fixture.attach",
>(
  causeCode: "DOMAIN_FIXTURE_LOCK_HELD" | "DOMAIN_FIXTURE_UNRESOLVED",
  commandIdentity: Identity,
) {
  return station(causes, {
    commandIdentity,
    causeCode,
    trigger: "Fixture refusal.",
    guidance: {
      handoff:
        commandIdentity === "fixture.approve" ? handoff : attachmentHandoff,
    },
    repairAction: "Inspect the fixture case.",
    retryDelayPolicy: { kind: "none" },
    reachability: "required",
    unreachableRationale: null,
    independentObservable: "fixture-effect-ledger",
  });
}
const textValue = (maxLength: number) =>
  ({
    kind: "string",
    minLength: 1,
    maxLength,
    choices: null,
    pattern: null,
  }) as const;
const noInput = {
  arguments: [],
  options: [],
  constraints: [],
  deadlineSupported: false,
  additionalArguments: false,
  additionalOptions: false,
} as const;

function inspectCommands() {
  const pending = command({
    identity: "fixture.pending",
    route: ["pending"],
    summary: "Inspect pending fixture work.",
    input: {
      ...noInput,
      options: [
        {
          name: "--all",
          aliases: [],
          summary: "All.",
          required: false,
          repeatable: false,
          value: null,
        },
      ],
    },
    capabilities: {},
    stations: [
      station(commonCauses, {
        commandIdentity: "fixture.pending",
        causeCode: "SUCCESS_UNCHANGED",
        trigger: "Read pending fixture work.",
        guidance: {
          next: { kind: "instruction", text: "Inspect pending fixture state." },
        },
        repairAction: null,
        retryDelayPolicy: { kind: "none" },
        reachability: "required",
        unreachableRationale: null,
        independentObservable: "fixture-pending-read",
      }),
    ],
    run: () => {
      throw new Error("Not exercised.");
    },
  });
  const inspectCase = command({
    identity: "fixture.case",
    route: ["case"],
    summary: "Inspect one fixture case.",
    input: {
      ...noInput,
      arguments: [
        {
          name: "operation",
          summary: "Case.",
          required: true,
          value: textValue(8),
        },
      ],
    },
    capabilities: {},
    run: () => {
      throw new Error("Not exercised.");
    },
  });
  return [pending, inspectCase] as const;
}

function kernelContract(place: World) {
  const lockHeld = stationFor("DOMAIN_FIXTURE_LOCK_HELD", "fixture.approve");
  const unresolved = stationFor("DOMAIN_FIXTURE_UNRESOLVED", "fixture.approve");
  const approve = changing({
    identity: "fixture.approve",
    route: ["approve"],
    summary: "Apply one approved fixture preview.",
    input: {
      ...noInput,
      arguments: [
        {
          name: "preview",
          summary: "Preview.",
          required: true,
          value: textValue(64),
        },
      ],
    },
    futureBindings: { preview: "preview" },
    capabilities: { approval: true },
    effectClass: "repository-local",
    effects: ["note.write"],
    ifNoResult: state.ifNoResult,
    heldLockCause: "DOMAIN_FIXTURE_LOCK_HELD",
    unresolvedCaseCause: "DOMAIN_FIXTURE_UNRESOLVED",
    stations: [lockHeld, unresolved],
    approval: (parsed) => ({
      authority: boundary.authority("owner-a"),
      resource: boundary.resource("resource-1"),
      resourceRevision: revision(place),
      effects: ["note.write"],
      preview: parsed.preview,
    }),
    run: (_parsed, context) => {
      emit({ phase: "handler" });
      return Promise.resolve(
        context.result(lockHeld, {
          message: "Fixture handler reached.",
          data: null,
          bindings: { preview: "p" },
          effects: {
            completed: [],
            remaining: [],
            uncertain: [],
            inventoryComplete: true,
          },
        }),
      );
    },
  });
  return contractFor({
    cli: "fixture",
    commands: [approve, ...inspectCommands()],
    causes,
    state,
    installedAddons: ["state", "approval"],
  });
}

async function kernel(map: Fields): Promise<void> {
  const place = world(map);
  const kernelUnderTest = kernelContract(place);
  const raws = map.get("invocations");
  if (!Array.isArray(raws)) throw new Error("Driver invocations.");
  const invocations: unknown[] = raws;
  for (const [index, argv] of invocations.entries()) {
    const runId = `kernel-${index}`;
    const { capability } = approval({
      operation: "fixture.approve",
      runId,
      store: { grants: place.grants, consumed: place.consumed },
      now: () => now(place),
    });
    const observed = {
      approval: {
        check(request: Parameters<typeof capability.check>[0]) {
          appendFileSync(place.contacts, `${runId}\n`);
          return capability.check(request);
        },
      },
    };
    const result = await kernelUnderTest.execute(
      "fixture.approve",
      argv,
      runId,
      observed,
    );
    emit({ phase: "kernel", index, result: result.ok ? { ok: true } : result });
  }
}

async function frontDoor(map: Fields): Promise<void> {
  const place = world(map);
  await runProcess(kernelContract(place), {
    addons: {
      approval: {
        attach: (run) => ({
          capability: approval({
            operation: "fixture.approve",
            runId: run.runId,
            store: { grants: place.grants, consumed: place.consumed },
            now: () => now(place),
          }).capability,
          settle: () => Promise.resolve(true),
          killNow: () => undefined,
        }),
      },
    },
  })(texts(map, "argv"));
}

async function whileAttached(
  runId: string,
  work: () => Promise<void>,
): Promise<void> {
  let failure: { readonly error: unknown } | undefined;
  const lockHeld = stationFor("DOMAIN_FIXTURE_LOCK_HELD", "fixture.attach");
  const attach = changing({
    identity: "fixture.attach",
    route: ["attach"],
    summary: "Run one fixture flow inside an attached run.",
    input: noInput,
    capabilities: {},
    effectClass: "repository-local",
    effects: ["note.write"],
    ifNoResult: state.ifNoResult,
    heldLockCause: "DOMAIN_FIXTURE_LOCK_HELD",
    unresolvedCaseCause: "DOMAIN_FIXTURE_UNRESOLVED",
    stations: [lockHeld, stationFor("DOMAIN_FIXTURE_UNRESOLVED", "fixture.attach")],
    run: async (_parsed, context) => {
      try {
        await work();
      } catch (error) {
        failure = { error };
      }
      return context.result(lockHeld, {
        message: "Fixture flow finished.",
        data: null,
        effects: {
          completed: [],
          remaining: [],
          uncertain: [],
          inventoryComplete: true,
        },
      });
    },
  });
  const executed = await contractFor({
    cli: "fixture",
    commands: [attach, ...inspectCommands()],
    causes,
    state,
    installedAddons: ["state"],
  }).execute("fixture.attach", [], runId);
  if (failure) throw failure.error;
  if (!executed.ok) throw new Error("Fixture run was not attached.");
}
function attached(settings: Common, work: () => Promise<void>) {
  const flow = () => whileAttached(settings.runId, work);
  return settings.coordination === "foreign-run"
    ? whileAttached(`${settings.runId}-other`, flow)
    : flow();
}

const mode = oneOf(input, "mode", [
  "hold",
  "apply",
  "authority",
  "kernel",
  "front-door",
]);
switch (mode) {
  case "hold":
    await hold(world(input));
    break;
  case "apply":
    await attached(common(input), () => apply(input));
    break;
  case "authority":
    await attached(common(input), () => authority(input));
    break;
  case "kernel":
    await kernel(input);
    break;
  case "front-door":
    await frontDoor(input);
    break;
}
process.exit(0);
