function seamState(): StateAddonDescriptor {
  return contractFor.stateAddon((issuer) => ({
    ifNoResult: issuer.ifNoResult({
      kind: "command",
      command: "kf.pending",
      args: [],
    }),
    attach<Effect extends string>(runId: string, effects: readonly Effect[]) {
      note("attach state");
      const attachBlock = Number(env.KF_STATE_ATTACH_BLOCK ?? "0");
      if (attachBlock > 0) block(attachBlock);
      if (env.KF_STATE_ATTACH === "throw")
        throw new Error("Fixture state attach fault.");
      return {
        async transaction(
          work: (transaction: EffectTransaction<Effect>) => Promise<void>,
        ) {
          const done = new Set<Effect>();
          const unsure = new Set<Effect>();
          await work({
            async record<T>(
              effect: Effect,
              operation: (capability: RecordedEffect<Effect>) => Promise<T>,
              readBack: (result: T) => Promise<boolean>,
            ) {
              const value = await operation(
                issuer.recordedEffect(runId, effect, (_run, _effect, start) =>
                  start(),
                ),
              );
              let confirmed = false;
              try {
                confirmed = await readBack(value);
              } catch {
                confirmed = false;
              }
              (confirmed ? done : unsure).add(effect);
              return value;
            },
          });
          return issuer.transactionProof(runId, observe(effects, done, unsure));
        },
      };
    },
  }));
}
function observe<Effect extends string>(
  declared: readonly Effect[],
  done: ReadonlySet<Effect>,
  unsure: ReadonlySet<Effect>,
): StateObservation<Effect> {
  const completedList = declared.filter((effect) => done.has(effect));
  const uncertain = declared.filter((effect) => unsure.has(effect));
  const remaining = declared.filter(
    (effect) => !done.has(effect) && !unsure.has(effect),
  );
  const [firstUnsure, ...moreUnsure] = uncertain;
  if (firstUnsure !== undefined)
    return {
      transactionState: "unknown",
      effects: {
        completed: completedList,
        remaining,
        uncertain: [firstUnsure, ...moreUnsure],
        inventoryComplete: true,
      },
    };
  const [firstDone, ...moreDone] = completedList;
  if (firstDone === undefined)
    return {
      transactionState: "unchanged",
      effects: { completed: [], remaining, uncertain: [], inventoryComplete: true },
    };
  const [firstRemaining, ...moreRemaining] = remaining;
  return firstRemaining === undefined
    ? {
        transactionState: "completed",
        effects: {
          completed: [firstDone, ...moreDone],
          remaining: [],
          uncertain: [],
          inventoryComplete: true,
        },
      }
    : {
        transactionState: "partially-completed",
        effects: {
          completed: [firstDone, ...moreDone],
          remaining: [firstRemaining, ...moreRemaining],
          uncertain: [],
          inventoryComplete: true,
        },
      };
}

function approvalAddon(operation: string): RunAddon<ApprovalCapability> {
  const mode = env.KF_APPROVAL ?? "grant";
  const settle = () => {
    note("settle approval");
    return Promise.resolve(true);
  };
  if (mode === "reject")
    return {
      attach: (() =>
        Promise.reject(
          new Error("Fixture approval attach rejection."),
        )) as unknown as RunAddon<ApprovalCapability>["attach"],
    };
  return {
    attach(run) {
      if (mode === "sync-throw")
        throw new Error("Fixture approval attach fault.");
      const shipped =
        mode === "shipped"
          ? approval({
              operation,
              runId: run.runId,
              store: {
                grants: env.KF_GRANTS ?? "",
                consumed: env.KF_GRANTS ?? "",
              },
            }).capability
          : undefined;
      note("attach approval");
      const checkBlock = Number(env.KF_CHECK_BLOCK ?? "0");
      return {
        capability: {
          check(request) {
            note("check-start");
            if (checkBlock > 0) block(checkBlock);
            return shipped ? shipped.check(request) : Promise.resolve(true);
          },
        },
        settle,
        killNow: () => undefined,
      };
    },
  };
}

function programsAddon(): RunAddon<ProgramsCapability> {
  if (env.KF_PROGRAMS_ATTACH === "reject")
    return {
      attach: (() =>
        Promise.reject(
          new Error("Fixture programs attach rejection."),
        )) as unknown as RunAddon<ProgramsCapability>["attach"],
    };
  return {
    attach(): AttachedAddon<ProgramsCapability> {
      note("attach programs");
      const started = () => {
        note("program-started");
        return Promise.resolve({
          stdout: "",
          stderr: "",
          exitCode: 0,
          terminationConfirmed: true,
          outputTruncated: false,
        });
      };
      return {
        capability: {
          inspect: started,
          supplied: started,
          domain: () => {
            throw new Error("Fixture programs add-on offers no domain program.");
          },
        },
        settle() {
          note("settle programs");
          if (env.KF_SETTLE === "throw") throw new Error("Fixture settle fault.");
          return Promise.resolve(env.KF_SETTLE !== "false");
        },
        killNow: () => undefined,
      };
    },
  };
}

const proofVariant = env.KF_PROOF ?? "completed";
function wideEffects(count: number): readonly [string, ...string[]] {
  const effects = Array.from(
    { length: count },
    (_, index) => `fx.wide.${String(index).padStart(2, "0")}.${"w".repeat(480)}`,
  );
  const [first, ...rest] = effects;
  if (first === undefined) throw new Error("No wide effects.");
  return [first, ...rest];
}
const variantEffects: Readonly<Record<string, readonly [string, ...string[]]>> = {
  completed: ["kf.e"],
  unchanged: ["kf.e"],
  unknown: ["kf.e"],
  partial: ["kf.e1", "kf.e2"],
  wide40: wideEffects(40),
  wide25: wideEffects(25),
};
async function variantWork(transaction: EffectTransaction<string>) {
  const record = (effect: string, readBack: () => Promise<boolean>) =>
    transaction.record(
      effect,
      () => {
        note(`effect-applied ${effect}`);
        return Promise.resolve();
      },
      readBack,
    );
  const confirmed = () => Promise.resolve(true);
  switch (proofVariant) {
    case "unchanged":
      return;
    case "unknown":
      await record("kf.e", () => Promise.reject(new Error("Read-back fault.")));
      return;
    case "partial":
      await record("kf.e1", confirmed);
      return;
    default:
      for (const effect of variantEffects[proofVariant] ?? []) {
        await transaction.record(effect, () => Promise.resolve(), confirmed);
        note(`effect-applied ${effect.slice(0, 10)}`);
      }
  }
}
type Transact = (
  state: unknown,
  work: (transaction: EffectTransaction<string>) => Promise<void>,
) => Promise<unknown>;
function proofApply(state: StateAddonDescriptor, transact: Transact) {
  const effects = variantEffects[proofVariant];
  if (!effects) throw new Error(`Unknown proof variant ${proofVariant}.`);
  const after = env.KF_AFTER ?? "throw";
  return changing({
    identity: "kf.apply",
    route: ["apply"],
    summary: "Apply the kf effects.",
    input,
    capabilities: {},
    effectClass: "repository-local",
    effects,
    ifNoResult: state.ifNoResult,
    heldLockCause: "KF_HELD",
    unresolvedCaseCause: "KF_PRIOR",
    stations: [
      ...coordination("kf.apply"),
      completed("kf.apply"),
      station(table, {
        ...words,
        commandIdentity: "kf.apply",
        causeCode: "KF_PARTIAL",
        trigger: "A known subset of the kf effects completed.",
        repairAction: "Inspect the kf effects.",
        guidance: {
          handoff: {
            owner: "operator",
            reason: "Inspect the kf effects.",
            inspect: [{ kind: "command", command: "kf.pending", args: [] }],
          },
        },
        independentObservable: "kf.apply:partial",
      }),
    ],
    run: async (_input, context) => {
      note("handler-entered");
      if (after === "before") throw new Error("Handler fault before work.");
      const proof = (await transact(context.state, variantWork)) as never as {
        readonly transactionState: string;
      };
      note(`proof ${proof.transactionState}`);
      const facts = { message: "Done.", data: null } as const;
      if (after === "throw") throw new Error("Handler fault after the proof.");
      if (after === "refusal")
        return context.result(
          { commandIdentity: "kf.apply", causeCode: "KF_HELD" },
          { ...facts, effects: empty },
        );
      if (after === "invalid")
        return context.result(
          { commandIdentity: "kf.apply", causeCode: "SUCCESS_COMPLETED" },
          { ...facts, data: Number.NaN, effects: empty } as never,
        );
      if (after === "select")
        return context.result(
          { commandIdentity: "kf.apply", causeCode: env.KF_SELECT } as never,
          { ...facts, transaction: proof, effects: empty } as never,
        );
      const selected =
        proof.transactionState === "completed"
          ? "SUCCESS_COMPLETED"
          : proof.transactionState === "partially-completed"
            ? "KF_PARTIAL"
            : "KF_HELD";
      return context.result(
        { commandIdentity: "kf.apply", causeCode: selected } as never,
        { ...facts, transaction: proof, effects: empty } as never,
      );
    },
  });
}

  };
}

const waitMilliseconds = Number(env.KF_WAIT ?? "0");
function programsApply(state: StateAddonDescriptor) {
  const effects =
    variantEffects[
      proofVariant === "none" || proofVariant === "stalled"
        ? "completed"
        : proofVariant
    ];
  if (!effects) throw new Error(`Unknown proof variant ${proofVariant}.`);
  return changing({
    identity: "kf.apply",
    route: ["apply"],
    summary: "Apply the kf effects and run one owned program.",
    input,
    capabilities: { programs: true },
    effectClass: "repository-local",
    effects,
    ifNoResult: state.ifNoResult,
    heldLockCause: "KF_HELD",
    unresolvedCaseCause: "KF_PRIOR",
    stations: [
      ...coordination("kf.apply"),
      completed("kf.apply"),
      station(table, {
        ...words,
        commandIdentity: "kf.apply",
        causeCode: "KF_PARTIAL",
        trigger: "A known subset of the kf effects completed.",
        repairAction: "Inspect the kf effects.",
        guidance: {
          handoff: {
            owner: "operator",
            reason: "Inspect the kf effects.",
            inspect: [{ kind: "command", command: "kf.pending", args: [] }],
          },
        },
        independentObservable: "kf.apply:partial",
      }),
    ],
    run: async (_input, context) => {
      note("handler-entered");
      if (proofVariant === "stalled")
        await context.state.transaction(async () => {
          await context.programs.inspect(program);
          await sleep(waitMilliseconds);
        });
      const proof =
        proofVariant === "none" || proofVariant === "stalled"
          ? undefined
          : await context.state.transaction(variantWork);
      await context.programs.inspect(program);
      if (waitMilliseconds > 0) await sleep(waitMilliseconds);
      note("handler-returned");
      const selected =
        proof?.transactionState === "completed"
          ? "SUCCESS_COMPLETED"
          : proof?.transactionState === "partially-completed"
            ? "KF_PARTIAL"
            : "KF_HELD";
      return context.result(
        { commandIdentity: "kf.apply", causeCode: selected } as never,
        {
          message: "Done.",
          data: null,
          effects: empty,
          ...(proof ? { transaction: proof } : {}),
        } as never,
      );
    },
  });
}

    if (waitMilliseconds > 0) await sleep(waitMilliseconds);

      {
        commandIdentity: "kf.watch",

      } as never,
      { message: "Watched.", data: null, effects: empty },
    );

}

function stepCommands() {
  const variant = env.KF_STEP ?? "next";
  const target =
    variant === "root" ? "kf.root" : variant === "deep" ? "kf.a" : "kf.list";
  const step = { kind: "command", command: target, args: [...stepArgs] } as const;
  const go = command({
    identity: "kf.go",
    route: ["go"],
    summary: "Go somewhere next.",
    input,
    capabilities: {},
    stations: [
      station(commonCauses, {
        ...words,
        commandIdentity: "kf.go",
        causeCode: "SUCCESS_UNCHANGED",
        trigger: "Went.",
        guidance: {
          next: variant === "inspect-entry" || variant === "if-no-result"
            ? { kind: "done" }
            : step,
        },
        repairAction: null,
        independentObservable: "kf.go:went",
      }),
      ...(variant === "inspect-entry"
        ? [
            station(table, {
              ...words,
              commandIdentity: "kf.go",
              causeCode: "KF_HELD",
              trigger: "Held.",
              repairAction: "Inspect the list.",
              guidance: {
                handoff: {
                  owner: "operator",
                  reason: "Inspect the list.",
                  inspect: [step],
                },
              },
              independentObservable: "kf.go:held",
            }),
          ]
        : []),
    ],
    run: (_input, context) => {
      note("handler-entered");
      return context.result(
        { commandIdentity: "kf.go", causeCode: "SUCCESS_UNCHANGED" },
        { message: "Went.", data: null, effects: empty },
      );
    },
  });
  const commands = [
    go,
    lister("kf.list", ["list"], ["name"]),
    lister("kf.list-all", ["list", "all"], []),
    ...(variant === "root" ? [lister("kf.root", [], ["name"])] : []),
    ...(variant === "deep"
      ? [
          lister("kf.a", ["a"], ["first", "second"]),
          lister("kf.a-b-c", ["a", "b", "c"], []),
        ]
      : []),
  ];
  if (variant !== "if-no-result")
    return contractFor({ cli: "kf", commands, causes } as never);
  const state = contractFor.stateAddon((issuer) => ({
    ifNoResult: issuer.ifNoResult({
      kind: "command",
      command: "kf.pending",
      args: [...stepArgs],
    }),
    attach() {
      throw new Error("No state execution in an admission fixture.");
    },
  }));
  const pendingList = command({
    identity: "kf.pending",
    route: ["pending"],
    summary: "Inspect pending kf changes.",
    input: { ...input, arguments: [operand("scope")] },
    capabilities: {},
    stations: [
      station(commonCauses, {
        ...words,
        commandIdentity: "kf.pending",
        causeCode: "SUCCESS_UNCHANGED",
        trigger: "Read pending.",
        guidance: {
          next: { kind: "instruction", text: "Inspect the pending kf changes." },
        },
        repairAction: null,
        independentObservable: "kf.pending:read",
      }),
    ],
    run: (_input, context) =>
      context.result(
        { commandIdentity: "kf.pending", causeCode: "SUCCESS_UNCHANGED" },
        { message: "Read pending.", data: null, effects: empty },
      ),
  });
  return contractFor({
    cli: "kf",
    commands: [
      ...commands,
      apply(state),
      pendingList,
      lister("kf.pending-all", ["pending", "all"], []),
    ],
    causes,
    state,
    installedAddons: ["state"],
  } as never);
}

const lateMode = env.KF_LATE ?? "transaction";
const lateMilliseconds = Number(env.KF_LATE_MS ?? "0");
function recordEffect(transaction: EffectTransaction<string>, effect: string) {
  return transaction.record(
    effect,
    () => {
      note(`effect-applied ${effect}`);
      return Promise.resolve();
    },
    () => Promise.resolve(true),
  );
}
function lateApply(state: StateAddonDescriptor) {
  const effects: readonly [string, ...string[]] =
    lateMode === "effect" || lateMode === "both" ? ["kf.e1", "kf.e2"] : ["kf.e"];
  return changing({
    identity: "kf.apply",
    route: ["apply"],
    summary: "Apply the kf effects late.",
    input,
    capabilities: {},
    effectClass: "repository-local",
    effects,
    ifNoResult: state.ifNoResult,
    heldLockCause: "KF_HELD",
    unresolvedCaseCause: "KF_PRIOR",
    stations: [...coordination("kf.apply"), completed("kf.apply")],
    run: async (_input, context) => {
      note("handler-entered");
      const refused = (error: unknown) => {
        note("transaction-refused");
        throw error;
      };
      if (lateMode === "transaction") {
        await sleep(lateMilliseconds);
        note("late-call");
      }
      const proof = await context.state
        .transaction(async (transaction) => {
          if (lateMode === "effect" || lateMode === "both")
            await recordEffect(transaction, "kf.e1");
          if (lateMode === "effect") await sleep(lateMilliseconds);
          for (const effect of lateMode === "effect" || lateMode === "both"
            ? ["kf.e2"]
            : ["kf.e"])
            await recordEffect(transaction, effect);
        })
        .catch(refused);
      if (lateMode === "early" || lateMode === "both") await sleep(2_000);
      return context.result(
        { commandIdentity: "kf.apply", causeCode: "SUCCESS_COMPLETED" } as never,
        { message: "Applied.", data: null, transaction: proof } as never,
      );
    },
  });
}
